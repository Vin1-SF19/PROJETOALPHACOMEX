it('N01: resposta de A remove o rascunho B e mostra Salvo antes de B ser enviado', async () => {
  let release!: (value: Awaited<ReturnType<typeof AtualizarNota>>) => void;
  vi.mocked(AtualizarNota).mockImplementationOnce(() => new Promise(resolve => { release=resolve; }));
  await mountNote(); await typeBody('A'); await act(async () => {await vi.advanceTimersByTimeAsync(1500);});
  expect(AtualizarNota).toHaveBeenCalledWith(expect.objectContaining({plainText:'A'}));
  await typeBody('B mais recente');
  expect(localStorage.getItem(getNotaRascunhoStorageKey('audit-note'))).toContain('B mais recente');
  await act(async () => {release({success:true,data:{id:'audit-note',title:'Nota fictícia',currentVersion:2,updatedAt:new Date()}});});
  expect(AtualizarNota).toHaveBeenCalledTimes(1);
  expect(host.querySelector('[data-status]')?.textContent).toBe('Salvo');
  expect(localStorage.getItem(getNotaRascunhoStorageKey('audit-note'))).toBeNull();
});
